@startuml
actor User

participant "main.py" as Main
participant "DatasetLoader" as Loader
participant "SyntheticDataGenerator" as Generator
participant "AttentionClassifier" as Classifier
participant "AttentionMonitoringSystem" as System
participant "FileSystem" as FS

User -> Main: Run main()
Main -> Loader: Initialize(dataset_path)
Loader -> Loader: load_images()
Loader -> Main: Return dataset

Main -> Generator: Initialize(seed=42)
Main -> Generator: generate_synthetic_data()
Generator -> Generator: Create synthetic data
Generator -> Main: Return X_synthetic, y_synthetic

Main -> Classifier: Initialize(model_type)
Main -> Classifier: train(X_synthetic, y_synthetic)
Classifier -> Classifier: Optimize hyperparameters
Classifier -> Classifier: Fit model
Classifier -> Main: Return accuracy

Main -> FS: Save best model (attention_classifier.pkl)

Main -> System: Initialize(classifier_model_path)
System -> System: Setup FaceDetector, FeatureExtractor, Classifier
System -> Main: Return system instance

Main -> System: analyze_dataset(dataset_loader)
System -> Loader: get_sample_images()
System -> System: Process frames
System -> Main: Return analysis results

Main -> FS: Save report (complete_analysis_results.json)

Main -> User: Prompt for webcam demo
User -> Main: Answer "y"
Main -> System: demo_real_time_monitoring()
System -> System: Process webcam frames
System -> FS: Save demo results
System -> Main: Return demo report

Main -> User: Display completion message

@enduml
